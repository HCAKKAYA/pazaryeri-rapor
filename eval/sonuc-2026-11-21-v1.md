```sql
SELECT urun, SUM(adet) FILTER (WHERE durum = 'iade') AS iade_adet, SUM(adet) FILTER (WHERE durum = 'iade') / NULLIF(SUM(adet) FILTER (WHERE durum IN ('teslim','iade')), 0) AS iade_orani FROM satis GROUP BY urun HAVING SUM(adet) FILTER (WHERE durum = 'iade') > 0 ORDER BY iade_adet DESC LIMIT 10
```
